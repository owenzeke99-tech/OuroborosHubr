local correctKey = "OuroborosHubKeyKalwowj - Kalapwownkeoeneje"
local fileName = "OuroborosHub_KeyData.json"

-- Hàm kiểm tra xem đã lưu key và còn hạn 24h không
local function checkSavedKey()
    if readfile and writefile and isfile then
        if isfile(fileName) then
            local success, data = pcall(function()
                return HttpService:JSONDecode(readfile(fileName))
            end)
            if success and data and data.key == correctKey then
                -- Kiểm tra thời hạn 24 giờ (86400 giây)
                if os.time() - data.time < 86400 then
                    return true
                end
            end
        end
    end
    return false
end

-- Hàm lưu key khi nhập đúng
local function saveKey()
    if writefile then
        pcall(function()
            local data = {
                key = correctKey,
                time = os.time()
            }
            writefile(fileName, HttpService:JSONEncode(data))
        end)
    end
end

-- Nếu đã lưu key hợp lệ trong vòng 24h thì tự động vào luôn, không hiện bảng key
if checkSavedKey() then
    loadstring(game:HttpGet("https://raw.githubusercontent.com/joustingmatch/Ouroboros/main/loader.lua"))()
    return
end

-- Hiệu ứng lắc ngang siêu mượt (Smooth Shake Animation)
local function shakeFrame()
    local originalPos = UDim2.new(0.5, -220, 0.5, -125)
    local info = TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
    
    local shake1 = TweenService:Create(MainFrame, info, {Position = originalPos + UDim2.new(0, 10, 0, 0)})
    local shake2 = TweenService:Create(MainFrame, info, {Position = originalPos + UDim2.new(0, -10, 0, 0)})
    local shake3 = TweenService:Create(MainFrame, info, {Position = originalPos + UDim2.new(0, 6, 0, 0)})
    local shake4 = TweenService:Create(MainFrame, info, {Position = originalPos + UDim2.new(0, -6, 0, 0)})
    local shake5 = TweenService:Create(MainFrame, info, {Position = originalPos})
    
    shake1:Play()
    shake1.Completed:Wait()
    shake2:Play()
    shake2.Completed:Wait()
    shake3:Play()
    shake3.Completed:Wait()
    shake4:Play()
    shake4.Completed:Wait()
    shake5:Play()
end

    else

        StatusLabel.Text = "Incorrect Key! Please try again."

        TextBox.Text = ""
        
        -- Chạy hiệu ứng lắc mượt
        task.spawn(shakeFrame)
    end
end)
